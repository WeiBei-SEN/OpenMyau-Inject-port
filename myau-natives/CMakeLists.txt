cmake_minimum_required(VERSION 3.20)
project(myau_native LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "A Windows x64 toolchain is required")
endif()
if(NOT DEFINED JAVA_INCLUDE)
    set(JAVA_INCLUDE "$ENV{JAVA_HOME}/include")
endif()
foreach(header jni.h jvmti.h)
    if(NOT EXISTS "${JAVA_INCLUDE}/${header}")
        message(FATAL_ERROR "Missing JDK header: ${JAVA_INCLUDE}/${header}")
    endif()
endforeach()
if(NOT DEFINED CLIENT_JAR)
    file(GLOB JAR_CANDIDATES "${CMAKE_CURRENT_SOURCE_DIR}/../OpenMyau-Inject/build/libs/*.jar")
    list(FILTER JAR_CANDIDATES EXCLUDE REGEX "(sources|javadoc|dev|without-deps)\\.jar$")
    list(LENGTH JAR_CANDIDATES JAR_COUNT)
    if(NOT JAR_COUNT EQUAL 1)
        message(FATAL_ERROR "Expected one client Jar in OpenMyau-Inject/build/libs; run Gradle first or set CLIENT_JAR explicitly")
    endif()
    list(GET JAR_CANDIDATES 0 CLIENT_JAR)
endif()
if(NOT EXISTS "${CLIENT_JAR}" OR IS_DIRECTORY "${CLIENT_JAR}")
    message(FATAL_ERROR "Client Jar does not exist: ${CLIENT_JAR}")
endif()
file(SIZE "${CLIENT_JAR}" CLIENT_JAR_SIZE)
if(CLIENT_JAR_SIZE LESS 22)
    message(FATAL_ERROR "Client Jar is empty or invalid")
endif()
get_filename_component(CLIENT_JAR "${CLIENT_JAR}" ABSOLUTE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Release")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Release")
add_library(myau_native SHARED dll/dllmain.cpp)
target_include_directories(myau_native PRIVATE dll/include "${JAVA_INCLUDE}")
set_target_properties(myau_native PROPERTIES PREFIX "")
set(PAYLOAD_DIR "${CMAKE_BINARY_DIR}/payload")
add_custom_command(OUTPUT "${PAYLOAD_DIR}/myau_native.dll"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${PAYLOAD_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:myau_native> "${PAYLOAD_DIR}/myau_native.dll"
    DEPENDS myau_native)
add_custom_target(myau_payload DEPENDS "${PAYLOAD_DIR}/myau_native.dll")
configure_file(loader/resources.rc.in "${CMAKE_BINARY_DIR}/resources.rc" @ONLY)
set_property(SOURCE "${CMAKE_BINARY_DIR}/resources.rc" PROPERTY OBJECT_DEPENDS "${CLIENT_JAR};${PAYLOAD_DIR}/myau_native.dll")
add_executable(myau_loader WIN32 loader/main.cpp loader/inject.cpp loader/target.cpp "${CMAKE_BINARY_DIR}/resources.rc")
add_dependencies(myau_loader myau_payload)
target_include_directories(myau_loader PRIVATE loader)
target_compile_definitions(myau_loader PRIVATE NOMINMAX UNICODE _UNICODE)
target_link_libraries(myau_loader PRIVATE d3d9 windowscodecs ole32 uuid user32 gdi32 shell32 psapi)
if(MINGW)
    foreach(target myau_native myau_loader)
        target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
        target_compile_options(${target} PRIVATE -Wall -Wextra)
    endforeach()
endif()
